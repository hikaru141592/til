    ```
    a = [3,4,5,6,7,8]
    s = 11
    arr = []
    (2 ** a.size).times do |i|
        subset = []
        a.size.times do |j|
            if i[j] == 1
                subset << a[j]
            end
        end
        if subset.sum == s
            arr << subset
        end
    end

    ```

  - 配列arrの各要素の合計値を求めるsum関数を定義する
    ```
    def sum(arr)
      if arr.empty?
        0
      else
        arr[0] + sum(arr[1..-1])
      end
    end
    ```
  - 処理高速化
    ```
    # 元のコード
    count = 0
    (7..7777777).each do |num|
        next if num % 7 != 0
        ketasu = num.to_s.length
        ketasu.times do |i|
            if num.to_s[i].to_i == 7
                count += 1
            end
        end
    end
    puts count

    # 改善後
    count = 0
    (7..7777777).step(7) do |num|
        count += num.to_s.count("7")
    end
    puts count
    ```
      - step(7)を用いることでループ回数自体を減らす
      - count("7")を用いることで内ループをなくす
